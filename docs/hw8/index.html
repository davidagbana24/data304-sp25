library(vegawidget)
library(vegabrite)
library(readr)
library(dplyr)
library(tidyr)

gas_url <- "https://calvin-data304.netlify.app/data/pump_price_for_gasoline_us_per_liter.csv"
countries_url <- "https://raw.githubusercontent.com/lukes/ISO-3166-Countries-with-Regional-Codes/master/all/all.csv"

gas <- read_csv(gas_url)
countries <- read_csv(countries_url)

# Converting to long, per gallon, join with countries
gas_long <- gas |>
  pivot_longer(-country, names_to = "year", values_to = "price_per_liter", values_transform = list(price_per_liter = as.numeric)) |>
  mutate(
    year = as.integer(year),
    price_per_gallon = price_per_liter * 3.785
  ) |>
  filter(!is.na(price_per_gallon))

# Joining on name
gas_joined <- gas_long |>
  left_join(countries, by = c("country" = "name"))

# Check unmatched countries
unmatched <- gas_joined |>
  filter(is.na(region)) |>
  distinct(country)
Code 
# Most recent year available per country
gas_2016 <- gas_joined |>
  filter(year == 2016, !is.na(region))

# Chart: price by country, colored by region, using alpha-3 code as label
gas_2016 |>
  vl_chart(width = 600, height = 400) |>
  vl_mark_point() |>
  vl_encode_x("price_per_gallon:Q") |>
  vl_axis_x(title = "Price per Gallon (USD)") |>
  vl_encode_y("region:N") |>
  vl_encode_color("region:N") |>
  vl_encode_tooltip_array(list("country", "alpha-3", "region", "price_per_gallon")) |>
  as_vegaspec()
